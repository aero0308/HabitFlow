    51→function greeting(name: string) {
    52→  const h = new Date().getHours();
    53→  if (h < 5) return `Burning the midnight oil, ${name}?`;
    54→  if (h < 12) return `Good morning, ${name}`;
    55→  if (h < 18) return `Good afternoon, ${name}`;
    56→  return `Good evening, ${name}`;
    57→}

    59→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    60→  const stroke = 12;
    61→  const r = (size - stroke) / 2;
    62→  const circ = 2 * Math.PI * r;
    63→  const offset = circ - (pct / 100) * circ;
    64→  return (
    65→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    66→      <svg width={size} height={size} className="-rotate-90">
    67→        <defs>
    68→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    69→            <stop offset="0%" stopColor="#10b981" />
    70→            <stop offset="100%" stopColor="#0d9488" />
    71→          </linearGradient>
    72→        </defs>
    73→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    74→        <motion.circle
    75→          cx={size / 2}
    76→          cy={size / 2}
    77→          r={r}
    78→          fill="none"
    79→          stroke="url(#ringGrad)"
    80→          strokeWidth={stroke}
    81→          strokeDasharray={circ}
    82→          strokeLinecap="round"
    83→          initial={{ strokeDashoffset: circ }}
    84→          animate={{ strokeDashoffset: offset }}
    85→          transition={{ duration: 1.2, ease: "easeOut" }}
    86→        />
    87→      </svg>
    88→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    89→        <motion.span
    90→          key={pct}
    91→          initial={{ scale: 0.5, opacity: 0 }}
    92→          animate={{ scale: 1, opacity: 1 }}
    93→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
    94→          className="text-4xl font-bold text-foreground tabular-nums"
    95→        >
    96→          {pct}%
    97→        </motion.span>
    98→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
    99→      </div>
   100→    </div>
   101→  );
   102→}

   104→function intensityClass(ratio: number, scheduled: number) {
   105→  if (scheduled === 0) return "bg-muted/30";
   106→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   107→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   108→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   109→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   110→  return "bg-emerald-600 dark:bg-emerald-500";
   111→}

   183→      // Number keys 1-9: toggle the Nth today's habit

   602→
