    41→function todayStr() {
    42→  const d = new Date();

    55→function freqLabel(frequency: string, customDays: number[]): string {
    56→  if (frequency === "daily") return "Every day";
    57→  if (frequency === "weekly") return "Weekly";
    58→  return `Custom · ${customDays.map((d) => WEEKDAYS[d]).join(", ") || "—"}`;
    59→}

    61→/** Build GitHub-style week columns from the flat 90-day calendar array. */
    62→function buildWeeks(
    63→  calendar: { date: string; scheduled: boolean; count: number; targetCount: number; completed: boolean }[],
    64→) {
    65→  if (calendar.length === 0) return { weeks: [] as (typeof calendar[number] | null)[][], monthMarkers: [] as { col: number; label: string }[] };
    66→
    67→  const first = calendar[0];
    68→  const firstMonFirst = (getDay(parseISO(first.date)) + 6) % 7;
    69→
    70→  // Flat list with leading null padding so the first column starts on Monday.
    71→  const padded: (typeof calendar[number] | null)[] = [
    72→    ...Array(firstMonFirst).fill(null),
    73→    ...calendar,
    74→  ];
    75→
    76→  // Group into 7-cell columns (each column = 1 week, Mon..Sun top→bottom).
    77→  const weeks: (typeof calendar[number] | null)[][] = [];
    78→  for (let i = 0; i < padded.length; i += 7) {
    79→    weeks.push(padded.slice(i, i + 7));
    80→  }
    81→
    82→  // Month markers: for each week column, find the first non-null entry and
    83→  // show its month abbreviation if it differs from the previous column's month.
    84→  const monthMarkers: { col: number; label: string }[] = [];
    85→  let lastMonth = "";
    86→  for (let col = 0; col < weeks.length; col++) {
    87→    const firstEntry = weeks[col].find((c) => c !== null);
    88→    if (!firstEntry) continue;
    89→    const month = format(parseISO(firstEntry.date), "MMM");
    90→    if (month !== lastMonth) {
    91→      monthMarkers.push({ col, label: month });
    92→      lastMonth = month;
    93→    }
    94→  }
    95→
    96→  return { weeks, monthMarkers };
    97→}

    99→function cellColor(
   100→  cell: { scheduled: boolean; completed: boolean } | null,
   101→): string {
   102→  if (!cell) return "bg-transparent";
   103→  if (!cell.scheduled) return "bg-muted/40 dark:bg-muted/30";
   104→  if (cell.completed) return "bg-emerald-600 dark:bg-emerald-500";
   105→  return "bg-emerald-200 dark:bg-emerald-900";
   106→}

   508→            </div>
   509→            <div>
   510→              <div className="font-semibold leading-tight">Day-of-week breakdown</div>
   511→              <div className="text-xs text-muted-foreground mt-0.5">Completion rate per weekday (all-time)</div>
   512→            </div>
   513→          </div>
   514→
   515→          <div className="mt-5 grid grid-cols-4 sm:grid-cols-7 gap-1.5 sm:gap-2">
   516→            {dowArr.map((d, i) => {
   517→              const ratio = d.scheduled === 0 ? 0 : d.completed / d.scheduled;
   518→              const isBest = d.day === bestDayIdx;
   519→              return (
   520→                <div
   521→                  key={i}
   522→                  className={cn(
   523→                    "rounded-lg p-1.5 sm:p-2 flex flex-col items-center justify-center text-center aspect-[3/4] sm:aspect-[3/4] transition-all",
   524→                    d.scheduled === 0
   525→                      ? "bg-muted/30"
   526→                      : ratio === 0
   527→                        ? "bg-muted/60 dark:bg-muted/40"
   528→                        : ratio < 0.34
   529→                          ? "bg-emerald-200 dark:bg-emerald-900"
   530→                          : ratio < 0.67
   531→                            ? "bg-emerald-400 dark:bg-emerald-700"
   532→                            : ratio < 1
   533→                              ? "bg-emerald-500 dark:bg-emerald-600"
   534→                              : "bg-emerald-600 dark:bg-emerald-500",
   535→                    isBest && "ring-2 ring-amber-400 ring-offset-1 ring-offset-background",
   536→                  )}
   537→                >
   538→                  <div className={cn("text-[9px] sm:text-[10px] font-medium uppercase tracking-wide", ratio >= 0.34 && d.scheduled > 0 ? "text-white" : "text-foreground")}>
   539→                    {WEEKDAYS[d.day]}
   540→                  </div>
   541→                  <div className={cn("text-xs sm:text-base font-bold mt-1", ratio >= 0.34 && d.scheduled > 0 ? "text-white" : "text-foreground")}>
   542→                    {d.scheduled === 0 ? "—" : `${d.pct}%`}
   543→                  </div>
   544→                  <div className={cn("text-[8px] sm:text-[9px] mt-0.5", ratio >= 0.34 && d.scheduled > 0 ? "text-white/80" : "text-muted-foreground")}>
   545→                    {d.scheduled === 0 ? "n/a" : `${d.completed}/${d.scheduled}`}
   546→                  </div>
   547→                  <div className="h-3 mt-1 flex items-center justify-center">
   548→                    {isBest && <Trophy className="w-3 h-3 text-amber-300" />}
   549→                  </div>
   550→                </div>
   551→              );
   552→            })}
   553→          </div>
   554→          <p className="text-[11px] text-muted-foreground mt-3">
   555→            Hover any cell for details. <span className="text-foreground font-medium">{WEEKDAY_LABELS_LONG[bestDayIdx] ?? "—"}</span> is your strongest day.
   556→          </p>
   557→        </Card>
   558→      </div>
   559→
   560→      {/* Recent check-ins */}
   561→      <Card className="p-6">
   562→        <div className="flex items-start gap-3">
   563→          <div className="w-9 h-9 rounded-lg bg-emerald-500/10 flex items-center justify-center flex-shrink-0">
   564→            <CheckCircle2 className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   565→          </div>
   566→          <div>
   567→            <div className="font-semibold leading-tight">Recent check-ins</div>
   568→            <div className="text-xs text-muted-foreground mt-0.5">Last {recentCheckins.length} entries</div>
   569→          </div>
   570→        </div>
   571→
   572→        {recentCheckins.length === 0 ? (
   573→          <div className="mt-4 py-8 text-center text-sm text-muted-foreground">
   574→            No check-ins yet. Mark this habit done to start your streak!
   575→          </div>
   576→        ) : (
   577→          <div className="mt-4 max-h-64 overflow-y-auto custom-scroll pr-2">
   578→            <div className="space-y-1.5 pr-2">
   579→              {recentCheckins
   580→                .slice()
   581→                .reverse()
   582→                .map((c) => {
   583→                  const target = habit.targetCount;
   584→                  const completed = c.count >= target;
   585→                  return (
   586→                    <div
   587→                      key={c.id}
   588→                      className="flex items-center gap-3 p-2.5 rounded-lg border border-border/60 hover:bg-muted/40 transition-colors"
   589→                    >
   590→                      <div className="w-10 h-10 rounded-lg bg-muted/50 flex flex-col items-center justify-center flex-shrink-0">
   591→                        <span className="text-[10px] font-medium leading-none text-muted-foreground uppercase">
   592→                          {format(parseISO(c.date), "MMM")}
   593→                        </span>
   594→                        <span className="text-sm font-bold leading-none mt-0.5">
   595→                          {format(parseISO(c.date), "d")}
   596→                        </span>
   597→                      </div>
   598→                      <div className="flex-1 min-w-0">
   599→                        <div className="text-sm font-medium">
   600→                          {format(parseISO(c.date), "EEEE, MMM d, yyyy")}
   601→                        </div>
   602→                        {c.note && (
   603→                          <div className="flex items-center gap-1 mt-0.5 text-xs text-muted-foreground">
   604→                            <StickyNote className="w-3 h-3 flex-shrink-0" />
   605→                            <span className="truncate">{c.note}</span>
   606→                          </div>
   607→                        )}
   608→                      </div>
   609→                      <Badge
   610→                        variant={completed ? "default" : "secondary"}
   611→                        className={cn(
   612→                          "text-[11px] py-0.5 px-2 tabular-nums",
   613→                          completed && "bg-emerald-600 hover:bg-emerald-700 text-white",
   614→                        )}
   615→                      >
   616→                        {c.count}/{target}
   617→                      </Badge>
   618→                    </div>
   619→                  );
   620→                })}
   621→            </div>
   622→          </div>
   623→        )}
   624→      </Card>
   625→    </div>
   626→  );
   627→}
