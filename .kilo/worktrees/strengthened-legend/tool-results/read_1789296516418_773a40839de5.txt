    57→function greeting(name: string) {
    58→  const h = new Date().getHours();
    59→  if (h < 5) return `Burning the midnight oil, ${name}?`;
    60→  if (h < 12) return `Good morning, ${name}`;
    61→  if (h < 18) return `Good afternoon, ${name}`;
    62→  return `Good evening, ${name}`;
    63→}

    65→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    66→  const stroke = 12;
    67→  const r = (size - stroke) / 2;
    68→  const circ = 2 * Math.PI * r;
    69→  const offset = circ - (pct / 100) * circ;
    70→  return (
    71→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    72→      <svg width={size} height={size} className="-rotate-90">
    73→        <defs>
    74→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    75→            <stop offset="0%" stopColor="#10b981" />
    76→            <stop offset="100%" stopColor="#0d9488" />
    77→          </linearGradient>
    78→        </defs>
    79→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    80→        <motion.circle
    81→          cx={size / 2}
    82→          cy={size / 2}
    83→          r={r}
    84→          fill="none"
    85→          stroke="url(#ringGrad)"
    86→          strokeWidth={stroke}
    87→          strokeDasharray={circ}
    88→          strokeLinecap="round"
    89→          initial={{ strokeDashoffset: circ }}
    90→          animate={{ strokeDashoffset: offset }}
    91→          transition={{ duration: 1.2, ease: "easeOut" }}
    92→        />
    93→      </svg>
    94→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    95→        <motion.span
    96→          key={pct}
    97→          initial={{ scale: 0.5, opacity: 0 }}
    98→          animate={{ scale: 1, opacity: 1 }}
    99→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
   100→          className="text-4xl font-bold text-foreground tabular-nums"
   101→        >
   102→          {pct}%
   103→        </motion.span>
   104→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
   105→      </div>
   106→    </div>
   107→  );
   108→}

   110→function intensityClass(ratio: number, scheduled: number) {
   111→  if (scheduled === 0) return "bg-muted/30";
   112→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   113→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   114→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   115→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   116→  return "bg-emerald-600 dark:bg-emerald-500";
   117→}

   480→                        uncheckMut.mutate({ habitId: h.id, date: today });
   481→                        toast("Undo", { description: `${h.name} marked incomplete` });
   482→                      } else {
   483→                        checkinMut.mutate({ habitId: h.id, date: today, count: h.targetCount });
   484→                        toast.success("Nice work!", { description: `${h.name} completed` });
   485→                      }
   486→                    }}
   487→                    onInc={() => {
   488→                      const next = Math.min(h.count + 1, h.targetCount);
   489→                      checkinMut.mutate({ habitId: h.id, date: today, count: next });
   490→                    }}
   491→                    onDec={() => {
   492→                      const next = Math.max(h.count - 1, 0);
   493→                      if (next === 0) {
   494→                        uncheckMut.mutate({ habitId: h.id, date: today });
   495→                      } else {
   496→                        checkinMut.mutate({ habitId: h.id, date: today, count: next });
   497→                      }
   498→                    }}
   499→                    onClick={() => go({ name: "habit", id: h.id })}
   500→                    onAddNote={() => openNoteDialog(h.id, h.name, h.note)}
   501→                    onFreeze={() => freezeMut.mutate({ habitId: h.id, date: today })}
   502→                    onUnfreeze={() => unfreezeMut.mutate({ habitId: h.id, date: today })}
   503→                  />
   504→                </motion.div>
   505→              ))}
   506→            </AnimatePresence>
   507→          </div>
   508→        ) : (
   509→          <Card className="p-8 text-center">
   510→            <div className="w-12 h-12 rounded-full bg-emerald-500/10 flex items-center justify-center mx-auto mb-3">
   511→              <Sparkles className="w-6 h-6 text-emerald-500" />
   512→            </div>
   513→            <p className="font-medium mb-1">No habits scheduled for today</p>
   514→            <p className="text-sm text-muted-foreground mb-4">Enjoy the day off, or create a new habit to track.</p>
   515→            <Button onClick={() => go({ name: "habits" })}>
   516→              <Plus className="w-4 h-4 mr-1" /> Create a habit
   517→            </Button>
   518→          </Card>
   519→        )}
   520→      </div>
   521→
   522→      {/* Quote + Calendar row */}
   523→      <div className="grid gap-4 lg:grid-cols-3">
   524→        {/* Motivational quote */}
   525→        <Card className="p-6 lg:col-span-1 bg-gradient-to-br from-emerald-500/5 via-teal-500/5 to-transparent border-emerald-500/20">
   526→          <Quote className="w-8 h-8 text-emerald-500/40 mb-3" />
   527→          <p className="text-sm font-medium leading-relaxed italic text-foreground">
   528→            &ldquo;{quote.text}&rdquo;
   529→          </p>
   530→          <p className="text-xs text-muted-foreground mt-3">— {quote.author}</p>
   531→        </Card>
   532→
   533→        {/* Calendar heatmap */}
   534→        <div className="lg:col-span-2">
   535→          <div className="flex items-center justify-between mb-3">
   536→            <h2 className="text-lg font-semibold flex items-center gap-2">
   537→              <CalendarIcon className="w-5 h-5" /> Calendar
   538→            </h2>
   539→            <div className="flex items-center gap-1">
   540→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setCalMonth(subMonths(calMonth, 1))}>
   541→                <ChevronLeft className="w-4 h-4" />
   542→              </Button>
   543→              <span className="text-sm font-medium w-28 text-center">{format(calMonth, "MMMM yyyy")}</span>
   544→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setCalMonth(addMonths(calMonth, 1))}>
   545→                <ChevronRight className="w-4 h-4" />
   546→              </Button>
   547→            </div>
   548→          </div>
   549→
   550→          <Card className="p-4">
   551→            {calendarGrid ? (
   552→              <div>
   553→                <div className="grid grid-cols-7 gap-1 mb-1">
   554→                  {WEEKDAYS.map((d) => (
   555→                    <div key={d} className="text-center text-[10px] font-semibold text-muted-foreground py-1 uppercase tracking-wide">
   556→                      {d}
   557→                    </div>
   558→                  ))}
   559→                </div>
   560→                <div className="grid grid-cols-7 gap-1">
   561→                  {calendarGrid.padded.map((date, i) => {
   562→                    if (!date) return <div key={i} />;
   563→                    const ds = format(date, "yyyy-MM-dd");
   564→                    const dayData = calendarGrid.dayMap.get(ds);
   565→                    const ratio = dayData?.ratio ?? 0;
   566→                    const scheduled = dayData?.scheduled ?? 0;
   567→                    return (
   568→                      <Popover key={ds}>
   569→                        <PopoverTrigger asChild>
   570→                          <button
   571→                            className={cn(
   572→                              "aspect-square rounded-md text-xs flex items-center justify-center transition-all hover:scale-110 hover:ring-2 hover:ring-emerald-400 hover:ring-offset-1 hover:ring-offset-background hover:z-10 relative",
   573→                              intensityClass(ratio, scheduled),
   574→                              isToday(date) && "ring-2 ring-emerald-500 ring-offset-1 ring-offset-background",
   575→                            )}
   576→                            aria-label={`${ds}: ${dayData?.completed ?? 0} of ${scheduled} done`}
   577→                          >
   578→                            <span className={cn("font-medium", ratio > 0.4 ? "text-white" : "text-foreground")}>
   579→                              {date.getDate()}
   580→                            </span>
   581→                          </button>
   582→                        </PopoverTrigger>
   583→                        <PopoverContent className="w-64 p-3" align="center">
   584→                          <div className="text-sm font-semibold mb-1">{format(date, "EEEE, MMM d")}</div>
   585→                          {scheduled === 0 ? (
   586→                            <p className="text-xs text-muted-foreground">No habits scheduled.</p>
   587→                          ) : (
   588→                            <>
   589→                              <div className="flex items-center justify-between mb-2">
   590→                                <p className="text-xs text-muted-foreground">
   591→                                  {dayData?.completed ?? 0} of {scheduled} completed
   592→                                </p>
   593→                                <Badge variant="secondary" className="text-[10px]">
   594→                                  {Math.round(ratio * 100)}%
   595→                                </Badge>
   596→                              </div>
   597→                              <div className="space-y-1.5">
   598→                                {dayData?.habits.map((h) => (
   599→                                  <div key={h.habitId} className="flex items-center gap-2 text-xs">
   600→                                    <span className="w-5 text-center">{h.icon}</span>
   601→                                    <span className="flex-1 truncate">{h.name}</span>
   602→                                    <span
   603→                                      className={cn(
   604→                                        "px-1.5 py-0.5 rounded text-[10px] font-medium",
   605→                                        h.completed ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900 dark:text-emerald-300" : "bg-muted text-muted-foreground",
   606→                                      )}
   607→                                    >
   608→                                      {h.count}/{h.targetCount}
   609→                                    </span>
   610→                                  </div>
   611→                                ))}
   612→                              </div>
   613→                            </>
   614→                          )}
   615→                        </PopoverContent>
   616→                      </Popover>
   617→                    );
   618→                  })}
   619→                </div>
   620→                <div className="flex items-center justify-between mt-3">
   621→                  <button
   622→                    onClick={() => go({ name: "history" })}
   623→                    className="text-xs text-muted-foreground hover:text-foreground transition-colors flex items-center gap-1"
   624→                  >
   625→                    View full history <ArrowRight className="w-3 h-3" />
   626→                  </button>
   627→                  <div className="flex items-center gap-1 text-[10px] text-muted-foreground">
   628→                    <span>Less</span>
   629→                    <div className="w-3 h-3 rounded bg-muted/60" />
   630→                    <div className="w-3 h-3 rounded bg-emerald-200 dark:bg-emerald-900" />
   631→                    <div className="w-3 h-3 rounded bg-emerald-400 dark:bg-emerald-700" />
   632→                    <div className="w-3 h-3 rounded bg-emerald-600 dark:bg-emerald-500" />
   633→                    <span>More</span>
   634→                  </div>
   635→                </div>
   636→              </div>
   637→            ) : (
   638→              <Skeleton className="h-64 w-full" />
   639→            )}
   640→          </Card>
   641→        </div>
   642→      </div>
   643→
   644→      {/* Keyboard shortcuts dialog */}
   645→      <Dialog open={showShortcuts} onOpenChange={setShowShortcuts}>
   646→        <DialogContent className="sm:max-w-md">
   647→          <DialogHeader>
   648→            <DialogTitle className="flex items-center gap-2">
   649→              <Keyboard className="w-5 h-5 text-emerald-500" />
   650→              Keyboard shortcuts
   651→            </DialogTitle>
   652→          </DialogHeader>
   653→          <div className="space-y-2 py-2">
   654→            <ShortcutRow keys={["1", "…", "9"]} desc="Toggle the Nth today's habit" />
   655→            <ShortcutRow keys={["?"]} desc="Open this help dialog" />
   656→            <ShortcutRow keys={["Esc"]} desc="Close dialogs" />
   657→          </div>
   658→          <div className="text-xs text-muted-foreground bg-muted/50 rounded-lg p-3">
   659→            Shortcuts only work on the dashboard when no input is focused.
   660→          </div>
   661→          <DialogFooter>
   662→            <Button onClick={() => setShowShortcuts(false)}>Got it</Button>
   663→          </DialogFooter>
   664→        </DialogContent>
   665→      </Dialog>
   666→
   667→      {/* Note dialog */}
   668→      <Dialog open={!!noteDialog} onOpenChange={(o) => !o && setNoteDialog(null)}>
   669→        <DialogContent className="sm:max-w-md">
   670→          <DialogHeader>
   671→            <DialogTitle className="flex items-center gap-2">
   672→              <StickyNote className="w-5 h-5 text-amber-500" />
   673→              Note for {noteDialog?.habitName}
   674→            </DialogTitle>
   675→          </DialogHeader>
   676→          <div className="space-y-2 py-2">
   677→            <Label htmlFor="note">Add a note about today&apos;s check-in</Label>
   678→            <Textarea
   679→              id="note"
   680→              placeholder="How did it go? Anything to remember?"
   681→              value={noteDraft}
   682→              onChange={(e) => setNoteDraft(e.target.value)}
   683→              rows={4}
   684→              maxLength={500}
   685→            />
   686→            <p className="text-xs text-muted-foreground">{noteDraft.length}/500 characters</p>
   687→          </div>
   688→          <DialogFooter>
   689→            <Button variant="outline" onClick={() => setNoteDialog(null)}>Cancel</Button>
   690→            <Button onClick={saveNote} disabled={checkinMut.isPending}>Save note</Button>
   691→          </DialogFooter>
   692→        </DialogContent>
   693→      </Dialog>
   694→
   695→      {/* Mood modal */}
   696→      <MoodModal open={showMoodModal} onOpenChange={setShowMoodModal} />
   697→    </div>
   698→  );
