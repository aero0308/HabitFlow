    52→function greeting(name: string) {
    53→  const h = new Date().getHours();
    54→  if (h < 5) return `Burning the midnight oil, ${name}?`;
    55→  if (h < 12) return `Good morning, ${name}`;
    56→  if (h < 18) return `Good afternoon, ${name}`;
    57→  return `Good evening, ${name}`;
    58→}

    60→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    61→  const stroke = 12;
    62→  const r = (size - stroke) / 2;
    63→  const circ = 2 * Math.PI * r;
    64→  const offset = circ - (pct / 100) * circ;
    65→  return (
    66→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    67→      <svg width={size} height={size} className="-rotate-90">
    68→        <defs>
    69→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    70→            <stop offset="0%" stopColor="#10b981" />
    71→            <stop offset="100%" stopColor="#0d9488" />
    72→          </linearGradient>
    73→        </defs>
    74→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    75→        <motion.circle
    76→          cx={size / 2}
    77→          cy={size / 2}
    78→          r={r}
    79→          fill="none"
    80→          stroke="url(#ringGrad)"
    81→          strokeWidth={stroke}
    82→          strokeDasharray={circ}
    83→          strokeLinecap="round"
    84→          initial={{ strokeDashoffset: circ }}
    85→          animate={{ strokeDashoffset: offset }}
    86→          transition={{ duration: 1.2, ease: "easeOut" }}
    87→        />
    88→      </svg>
    89→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    90→        <motion.span
    91→          key={pct}
    92→          initial={{ scale: 0.5, opacity: 0 }}
    93→          animate={{ scale: 1, opacity: 1 }}
    94→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
    95→          className="text-4xl font-bold text-foreground tabular-nums"
    96→        >
    97→          {pct}%
    98→        </motion.span>
    99→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
   100→      </div>
   101→    </div>
   102→  );
   103→}

   105→function intensityClass(ratio: number, scheduled: number) {
   106→  if (scheduled === 0) return "bg-muted/30";
   107→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   108→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   109→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   110→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   111→  return "bg-emerald-600 dark:bg-emerald-500";
   112→}

   180→      const target = e.target as HTMLElement;
   181→      if (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable) return;
   182→      if (noteDialog || showShortcuts) return;
   183→
   184→      // Number keys 1-9: toggle the Nth today's habit
   185→      if (/^[1-9]$/.test(e.key)) {
   186→        const idx = Number(e.key) - 1;

   188→        if (habits && idx < habits.length) {

   197→      }
