    41→function todayStr() {
    42→  const d = new Date();
    43→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;

    54→function freqLabel(frequency: string, customDays: number[]): string {
    55→  if (frequency === "daily") return "Every day";
    56→  if (frequency === "weekly") return "Weekly";
    57→  return `Custom · ${customDays.map((d) => WEEKDAYS[d]).join(", ") || "—"}`;
    58→}

    60→/** Build GitHub-style week columns from the flat 90-day calendar array. */
    61→function buildWeeks(
    62→  calendar: { date: string; scheduled: boolean; count: number; targetCount: number; completed: boolean }[],
    63→) {
    64→  if (calendar.length === 0) return { weeks: [] as (typeof calendar[number] | null)[][], monthMarkers: [] as { col: number; label: string }[] };
    65→
    66→  const first = calendar[0];
    67→  const firstMonFirst = (getDay(parseISO(first.date)) + 6) % 7;
    68→
    69→  // Flat list with leading null padding so the first column starts on Monday.
    70→  const padded: (typeof calendar[number] | null)[] = [
    71→    ...Array(firstMonFirst).fill(null),
    72→    ...calendar,
    73→  ];
    74→
    75→  // Group into 7-cell columns (each column = 1 week, Mon..Sun top→bottom).
    76→  const weeks: (typeof calendar[number] | null)[][] = [];
    77→  for (let i = 0; i < padded.length; i += 7) {
    78→    weeks.push(padded.slice(i, i + 7));
    79→  }
    80→
    81→  // Month markers: for each week column, find the first non-null entry and
    82→  // show its month abbreviation if it differs from the previous column's month.
    83→  const monthMarkers: { col: number; label: string }[] = [];
    84→  let lastMonth = "";
    85→  for (let col = 0; col < weeks.length; col++) {
    86→    const firstEntry = weeks[col].find((c) => c !== null);
    87→    if (!firstEntry) continue;
    88→    const month = format(parseISO(firstEntry.date), "MMM");
    89→    if (month !== lastMonth) {
    90→      monthMarkers.push({ col, label: month });
    91→      lastMonth = month;
    92→    }
    93→  }
    94→
    95→  return { weeks, monthMarkers };
    96→}

    98→function cellColor(
    99→  cell: { scheduled: boolean; completed: boolean } | null,
   100→): string {
   101→  if (!cell) return "bg-transparent";
   102→  if (!cell.scheduled) return "bg-muted/40 dark:bg-muted/30";
   103→  if (cell.completed) return "bg-emerald-600 dark:bg-emerald-500";
   104→  return "bg-emerald-200 dark:bg-emerald-900";
   105→}

   513→            <CheckCircle2 className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   514→          </div>
   515→          <div>
   516→            <div className="font-semibold leading-tight">Recent check-ins</div>
   517→            <div className="text-xs text-muted-foreground mt-0.5">Last {recentCheckins.length} entries</div>
   518→          </div>
   519→        </div>
   520→
   521→        {recentCheckins.length === 0 ? (
   522→          <div className="mt-4 py-8 text-center text-sm text-muted-foreground">
   523→            No check-ins yet. Mark this habit done to start your streak!
   524→          </div>
   525→        ) : (
   526→          <ScrollArea className="mt-4 max-h-64 pr-3">
   527→            <div className="space-y-1.5 pr-2">
   528→              {recentCheckins
   529→                .slice()
   530→                .reverse()
   531→                .map((c) => {
   532→                  const target = habit.targetCount;
   533→                  const completed = c.count >= target;
   534→                  return (
   535→                    <div
   536→                      key={c.id}
   537→                      className="flex items-center gap-3 p-2.5 rounded-lg border border-border/60 hover:bg-muted/40 transition-colors"
   538→                    >
   539→                      <div className="w-10 h-10 rounded-lg bg-muted/50 flex flex-col items-center justify-center flex-shrink-0">
   540→                        <span className="text-[10px] font-medium leading-none text-muted-foreground uppercase">
   541→                          {format(parseISO(c.date), "MMM")}
   542→                        </span>
   543→                        <span className="text-sm font-bold leading-none mt-0.5">
   544→                          {format(parseISO(c.date), "d")}
   545→                        </span>
   546→                      </div>
   547→                      <div className="flex-1 min-w-0">
   548→                        <div className="text-sm font-medium">
   549→                          {format(parseISO(c.date), "EEEE, MMM d, yyyy")}
   550→                        </div>
   551→                        {c.note && (
   552→                          <div className="flex items-center gap-1 mt-0.5 text-xs text-muted-foreground">
   553→                            <StickyNote className="w-3 h-3 flex-shrink-0" />
   554→                            <span className="truncate">{c.note}</span>
   555→                          </div>
   556→                        )}
   557→                      </div>
   558→                      <Badge
   559→                        variant={completed ? "default" : "secondary"}
   560→                        className={cn(
   561→                          "text-[11px] py-0.5 px-2 tabular-nums",
   562→                          completed && "bg-emerald-600 hover:bg-emerald-700 text-white",
   563→                        )}
   564→                      >
   565→                        {c.count}/{target}
   566→                      </Badge>
   567→                    </div>
   568→                  );
   569→                })}
   570→            </div>
   571→          </ScrollArea>
   572→        )}
   573→      </Card>
   574→    </div>
   575→  );
   576→}
