    58→    58→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    59→    59→  const stroke = 12;
    60→    60→  const r = (size - stroke) / 2;
    61→    61→  const circ = 2 * Math.PI * r;
    62→    62→  const offset = circ - (pct / 100) * circ;
    63→    63→  return (
    64→    64→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    65→    65→      <svg width={size} height={size} className="-rotate-90">
    66→    66→        <defs>
    67→    67→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    68→    68→            <stop offset="0%" stopColor="#10b981" />
    69→    69→            <stop offset="100%" stopColor="#0d9488" />
    70→    70→          </linearGradient>
    71→    71→        </defs>
    72→    72→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    73→    73→        <motion.circle
    74→    74→          cx={size / 2}
    75→    75→          cy={size / 2}
    76→    76→          r={r}
    77→    77→          fill="none"
    78→    78→          stroke="url(#ringGrad)"
    79→    79→          strokeWidth={stroke}
    80→    80→          strokeDasharray={circ}
    81→    81→          strokeLinecap="round"
    82→    82→          initial={{ strokeDashoffset: circ }}
    83→    83→          animate={{ strokeDashoffset: offset }}
    84→    84→          transition={{ duration: 1.2, ease: "easeOut" }}
    85→    85→        />
    86→    86→      </svg>
    87→    87→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    88→    88→        <motion.span
    89→    89→          key={pct}
    90→    90→          initial={{ scale: 0.5, opacity: 0 }}
    91→    91→          animate={{ scale: 1, opacity: 1 }}
    92→    92→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
    93→    93→          className="text-4xl font-bold text-foreground tabular-nums"
    94→    94→        >
    95→    95→          {pct}%
    96→    96→        </motion.span>
    97→    97→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
    98→    98→      </div>
    99→    99→    </div>
   100→   100→  );
   101→   101→}

   103→   103→function intensityClass(ratio: number, scheduled: number) {
   104→   104→  if (scheduled === 0) return "bg-muted/30";
   105→   105→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   106→   106→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   107→   107→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   108→   108→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   109→   109→  return "bg-emerald-600 dark:bg-emerald-500";
   110→   110→}
