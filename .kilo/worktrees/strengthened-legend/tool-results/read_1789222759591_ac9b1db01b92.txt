    54→function greeting(name: string) {
    55→  const h = new Date().getHours();
    56→  if (h < 5) return `Burning the midnight oil, ${name}?`;
    57→  if (h < 12) return `Good morning, ${name}`;
    58→  if (h < 18) return `Good afternoon, ${name}`;
    59→  return `Good evening, ${name}`;
    60→}

    62→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    63→  const stroke = 12;
    64→  const r = (size - stroke) / 2;
    65→  const circ = 2 * Math.PI * r;
    66→  const offset = circ - (pct / 100) * circ;
    67→  return (
    68→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    69→      <svg width={size} height={size} className="-rotate-90">
    70→        <defs>
    71→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    72→            <stop offset="0%" stopColor="#10b981" />
    73→            <stop offset="100%" stopColor="#0d9488" />
    74→          </linearGradient>
    75→        </defs>
    76→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    77→        <motion.circle
    78→          cx={size / 2}
    79→          cy={size / 2}
    80→          r={r}
    81→          fill="none"
    82→          stroke="url(#ringGrad)"
    83→          strokeWidth={stroke}
    84→          strokeDasharray={circ}
    85→          strokeLinecap="round"
    86→          initial={{ strokeDashoffset: circ }}
    87→          animate={{ strokeDashoffset: offset }}
    88→          transition={{ duration: 1.2, ease: "easeOut" }}
    89→        />
    90→      </svg>
    91→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    92→        <motion.span
    93→          key={pct}
    94→          initial={{ scale: 0.5, opacity: 0 }}
    95→          animate={{ scale: 1, opacity: 1 }}
    96→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
    97→          className="text-4xl font-bold text-foreground tabular-nums"
    98→        >
    99→          {pct}%
   100→        </motion.span>
   101→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
   102→      </div>
   103→    </div>
   104→  );
   105→}

   107→function intensityClass(ratio: number, scheduled: number) {
   108→  if (scheduled === 0) return "bg-muted/30";
   109→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   110→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   111→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   112→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   113→  return "bg-emerald-600 dark:bg-emerald-500";
   114→}

   183→      const target = e.target as HTMLElement;

   185→      if (noteDialog || showShortcuts) return;
   186→
   187→      // Number keys 1-9: toggle the Nth today's habit

   189→        const idx = Number(e.key) - 1;

   200→      }

   477→        </Card>

   489→              <span className="text-sm font-medium w-28 text-center">{format(calMonth, "MMMM yyyy")}</span>

   491→                <ChevronRight className="w-4 h-4" />
   492→              </Button>
   493→            </div>
   494→          </div>
   495→
   496→          <Card className="p-4">

   499→                <div className="grid grid-cols-7 gap-1 mb-1">

   503→                    </div>
   504→                  ))}
   505→                </div>
   506→                <div className="grid grid-cols-7 gap-1">
   507→                  {calendarGrid.padded.map((date, i) => {
   508→                    if (!date) return <div key={i} />;
   509→                    const ds = format(date, "yyyy-MM-dd");
   510→                    const dayData = calendarGrid.dayMap.get(ds);
   511→                    const ratio = dayData?.ratio ?? 0;
   512→                    const scheduled = dayData?.scheduled ?? 0;
   513→                    return (
   514→                      <Popover key={ds}>
   515→                        <PopoverTrigger asChild>

   517→                            className={cn(
   518→                              "aspect-square rounded-md text-xs flex items-center justify-center transition-all hover:scale-110 hover:ring-2 hover:ring-emerald-400 hover:ring-offset-1 hover:ring-offset-background hover:z-10 relative",
   519→                              intensityClass(ratio, scheduled),
   520→                              isToday(date) && "ring-2 ring-emerald-500 ring-offset-1 ring-offset-background",
   521→                            )}
   522→                            aria-label={`${ds}: ${dayData?.completed ?? 0} of ${scheduled} done`}
   523→                          >
   524→                            <span className={cn("font-medium", ratio > 0.4 ? "text-white" : "text-foreground")}>
   525→                              {date.getDate()}
   526→                            </span>
   527→                          </button>
   528→                        </PopoverTrigger>
   529→                        <PopoverContent className="w-64 p-3" align="center">
   530→                          <div className="text-sm font-semibold mb-1">{format(date, "EEEE, MMM d")}</div>
   531→                          {scheduled === 0 ? (
   532→                            <p className="text-xs text-muted-foreground">No habits scheduled.</p>
   533→                          ) : (
   534→                            <>
   535→                              <div className="flex items-center justify-between mb-2">
   536→                                <p className="text-xs text-muted-foreground">
   537→                                  {dayData?.completed ?? 0} of {scheduled} completed
   538→                                </p>
   539→                                <Badge variant="secondary" className="text-[10px]">
   540→                                  {Math.round(ratio * 100)}%
   541→                                </Badge>
   542→                              </div>
   543→                              <div className="space-y-1.5">
   544→                                {dayData?.habits.map((h) => (
   545→                                  <div key={h.habitId} className="flex items-center gap-2 text-xs">
   546→                                    <span className="w-5 text-center">{h.icon}</span>
   547→                                    <span className="flex-1 truncate">{h.name}</span>
   548→                                    <span
   549→                                      className={cn(
   550→                                        "px-1.5 py-0.5 rounded text-[10px] font-medium",
   551→                                        h.completed ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900 dark:text-emerald-300" : "bg-muted text-muted-foreground",
   552→                                      )}
   553→                                    >
   554→                                      {h.count}/{h.targetCount}
   555→                                    </span>
   556→                                  </div>
   557→                                ))}
   558→                              </div>
   559→                            </>
   560→                          )}
   561→                        </PopoverContent>
   562→                      </Popover>
   563→                    );
   564→                  })}
   565→                </div>
   566→                <div className="flex items-center justify-between mt-3">
   567→                  <button
   568→                    onClick={() => go({ name: "history" })}
   569→                    className="text-xs text-muted-foreground hover:text-foreground transition-colors flex items-center gap-1"
   570→                  >
   571→                    View full history <ArrowRight className="w-3 h-3" />
   572→                  </button>
   573→                  <div className="flex items-center gap-1 text-[10px] text-muted-foreground">
   574→                    <span>Less</span>
   575→                    <div className="w-3 h-3 rounded bg-muted/60" />
   576→                    <div className="w-3 h-3 rounded bg-emerald-200 dark:bg-emerald-900" />
   577→                    <div className="w-3 h-3 rounded bg-emerald-400 dark:bg-emerald-700" />
   578→                    <div className="w-3 h-3 rounded bg-emerald-600 dark:bg-emerald-500" />
   579→                    <span>More</span>
   580→                  </div>
   581→                </div>
   582→              </div>
   583→            ) : (
   584→              <Skeleton className="h-64 w-full" />
   585→            )}
   586→          </Card>
   587→        </div>
   588→      </div>
   589→
   590→      {/* Keyboard shortcuts dialog */}
   591→      <Dialog open={showShortcuts} onOpenChange={setShowShortcuts}>
   592→        <DialogContent className="sm:max-w-md">
   593→          <DialogHeader>
   594→            <DialogTitle className="flex items-center gap-2">
   595→              <Keyboard className="w-5 h-5 text-emerald-500" />
   596→              Keyboard shortcuts
   597→            </DialogTitle>
   598→          </DialogHeader>
   599→          <div className="space-y-2 py-2">
   600→            <ShortcutRow keys={["1", "…", "9"]} desc="Toggle the Nth today's habit" />
   601→            <ShortcutRow keys={["?"]} desc="Open this help dialog" />
   602→            <ShortcutRow keys={["Esc"]} desc="Close dialogs" />
   603→          </div>
   604→          <div className="text-xs text-muted-foreground bg-muted/50 rounded-lg p-3">
   605→            Shortcuts only work on the dashboard when no input is focused.
   606→          </div>
   607→          <DialogFooter>
   608→            <Button onClick={() => setShowShortcuts(false)}>Got it</Button>
   609→          </DialogFooter>
   610→        </DialogContent>
   611→      </Dialog>
   612→
   613→      {/* Note dialog */}
   614→      <Dialog open={!!noteDialog} onOpenChange={(o) => !o && setNoteDialog(null)}>
   615→        <DialogContent className="sm:max-w-md">
   616→          <DialogHeader>
   617→            <DialogTitle className="flex items-center gap-2">
   618→              <StickyNote className="w-5 h-5 text-amber-500" />
   619→              Note for {noteDialog?.habitName}
   620→            </DialogTitle>
   621→          </DialogHeader>
   622→          <div className="space-y-2 py-2">
   623→            <Label htmlFor="note">Add a note about today&apos;s check-in</Label>
   624→            <Textarea
   625→              id="note"
   626→              placeholder="How did it go? Anything to remember?"
   627→              value={noteDraft}
   628→              onChange={(e) => setNoteDraft(e.target.value)}
   629→              rows={4}
   630→              maxLength={500}
   631→            />
   632→            <p className="text-xs text-muted-foreground">{noteDraft.length}/500 characters</p>
   633→          </div>
   634→          <DialogFooter>
   635→            <Button variant="outline" onClick={() => setNoteDialog(null)}>Cancel</Button>
   636→            <Button onClick={saveNote} disabled={checkinMut.isPending}>Save note</Button>
   637→          </DialogFooter>
   638→        </DialogContent>
   639→      </Dialog>
   640→
   641→      {/* Mood modal */}
   642→      <MoodModal open={showMoodModal} onOpenChange={setShowMoodModal} />
   643→    </div>
   644→  );

   701→function HabitRow({
   702→  habit,
   703→  index,
   704→  onToggle,
   705→  onInc,
   706→  onDec,
   707→  onClick,
   708→  onAddNote,
   709→  onFreeze,
   710→  onUnfreeze,
   711→}: {
   712→  habit: {
   713→    id: string;
   714→    name: string;
   715→    description: string;
   716→    color: string;
   717→    icon: string;
   718→    targetCount: number;
   719→    count: number;
   720→    completed: boolean;
   721→    frozen: boolean;
   722→    note: string;
   723→    streak: { current: number; longest: number };
   724→  };
   725→  index: number;
   726→  today: string;
   727→  onToggle: () => void;
   728→  onInc: () => void;
   729→  onDec: () => void;

   738→    <Card className={cn(
   739→      "p-3 sm:p-4 transition-all hover:shadow-md group",
   740→      habit.completed && "border-emerald-500/50 bg-emerald-500/5",
   741→      habit.frozen && !habit.completed && "border-sky-500/50 bg-sky-500/5",
   742→    )}>

   744→        {/* Keyboard shortcut number badge (hidden on mobile) */}

   746→          {index + 1}
   747→        </div>
   748→        <button
   749→          onClick={onClick}
   750→          className="w-10 h-10 rounded-lg flex items-center justify-center text-xl flex-shrink-0 transition-transform group-hover:scale-105"
   751→          style={{ backgroundColor: habit.color + "20" }}
   752→          aria-label={habit.name}
   753→        >
   754→          {habit.icon}
   755→        </button>
   756→

   760→            <span className="font-medium text-sm truncate">{habit.name}</span>

   762→              <CheckCircle2 className="w-3.5 h-3.5 text-emerald-500 flex-shrink-0" />

   766→                <Snowflake className="w-3 h-3" /> Frozen
   767→              </span>
   768→            )}

   770→              <StickyNote className="w-3 h-3 text-amber-500 flex-shrink-0" aria-label="Has note" />
   771→            )}
   772→          </div>
   773→          <div className="flex items-center gap-2 mt-1">

   775→              <Badge variant="secondary" className="text-[10px] py-0 px-1.5 h-4 gap-0.5">
   776→                <Flame className="w-2.5 h-2.5 text-orange-500" />

   778→              </Badge>
   779→            )}

   781→              <div className="flex items-center gap-1.5 flex-1 max-w-32">
   782→                <Progress value={progressPct} className="h-1.5" />

   827→                <>
   828→                  <CheckCircle2 className="w-4 h-4 mr-1" /> Done

   831→                "Mark done"

   833→            </Button>

   835→        </div>
   836→      </div>
   837→    </Card>
