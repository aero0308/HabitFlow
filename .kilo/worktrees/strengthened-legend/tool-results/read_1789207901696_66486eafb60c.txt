    60→    60→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    61→    61→  const stroke = 12;
    62→    62→  const r = (size - stroke) / 2;
    63→    63→  const circ = 2 * Math.PI * r;
    64→    64→  const offset = circ - (pct / 100) * circ;
    65→    65→  return (
    66→    66→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    67→    67→      <svg width={size} height={size} className="-rotate-90">
    68→    68→        <defs>
    69→    69→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    70→    70→            <stop offset="0%" stopColor="#10b981" />
    71→    71→            <stop offset="100%" stopColor="#0d9488" />
    72→    72→          </linearGradient>
    73→    73→        </defs>
    74→    74→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    75→    75→        <motion.circle
    76→    76→          cx={size / 2}
    77→    77→          cy={size / 2}
    78→    78→          r={r}
    79→    79→          fill="none"
    80→    80→          stroke="url(#ringGrad)"
    81→    81→          strokeWidth={stroke}
    82→    82→          strokeDasharray={circ}
    83→    83→          strokeLinecap="round"
    84→    84→          initial={{ strokeDashoffset: circ }}
    85→    85→          animate={{ strokeDashoffset: offset }}
    86→    86→          transition={{ duration: 1.2, ease: "easeOut" }}
    87→    87→        />
    88→    88→      </svg>
    89→    89→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    90→    90→        <motion.span
    91→    91→          key={pct}
    92→    92→          initial={{ scale: 0.5, opacity: 0 }}
    93→    93→          animate={{ scale: 1, opacity: 1 }}
    94→    94→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
    95→    95→          className="text-4xl font-bold text-foreground tabular-nums"
    96→    96→        >
    97→    97→          {pct}%
    98→    98→        </motion.span>
    99→    99→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
   100→   100→      </div>
   101→   101→    </div>
   102→   102→  );
   103→   103→}

   105→   105→function intensityClass(ratio: number, scheduled: number) {
   106→   106→  if (scheduled === 0) return "bg-muted/30";
   107→   107→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   108→   108→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   109→   109→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   110→   110→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   111→   111→  return "bg-emerald-600 dark:bg-emerald-500";
   112→   112→}
