    50→function greeting(name: string) {
    51→  const h = new Date().getHours();
    52→  if (h < 5) return `Burning the midnight oil, ${name}?`;
    53→  if (h < 12) return `Good morning, ${name}`;
    54→  if (h < 18) return `Good afternoon, ${name}`;
    55→  return `Good evening, ${name}`;
    56→}

    58→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    59→  const stroke = 12;
    60→  const r = (size - stroke) / 2;
    61→  const circ = 2 * Math.PI * r;
    62→  const offset = circ - (pct / 100) * circ;
    63→  return (
    64→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    65→      <svg width={size} height={size} className="-rotate-90">
    66→        <defs>
    67→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    68→            <stop offset="0%" stopColor="#10b981" />
    69→            <stop offset="100%" stopColor="#0d9488" />
    70→          </linearGradient>
    71→        </defs>
    72→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    73→        <motion.circle
    74→          cx={size / 2}
    75→          cy={size / 2}
    76→          r={r}
    77→          fill="none"
    78→          stroke="url(#ringGrad)"
    79→          strokeWidth={stroke}
    80→          strokeDasharray={circ}
    81→          strokeLinecap="round"
    82→          initial={{ strokeDashoffset: circ }}
    83→          animate={{ strokeDashoffset: offset }}
    84→          transition={{ duration: 1.2, ease: "easeOut" }}
    85→        />
    86→      </svg>
    87→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    88→        <motion.span
    89→          key={pct}
    90→          initial={{ scale: 0.5, opacity: 0 }}
    91→          animate={{ scale: 1, opacity: 1 }}
    92→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
    93→          className="text-4xl font-bold text-foreground tabular-nums"
    94→        >
    95→          {pct}%
    96→        </motion.span>
    97→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
    98→      </div>
    99→    </div>
   100→  );
   101→}

   103→function intensityClass(ratio: number, scheduled: number) {
   104→  if (scheduled === 0) return "bg-muted/30";
   105→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   106→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   107→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   108→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   109→  return "bg-emerald-600 dark:bg-emerald-500";
   110→}

   178→      if (noteDialog || showShortcuts) return;
   179→
   180→      // Number keys 1-9: toggle the Nth today's habit

   182→        const idx = Number(e.key) - 1;

   184→        if (habits && idx < habits.length) {
   185→          e.preventDefault();

   475→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setCalMonth(addMonths(calMonth, 1))}>
   476→                <ChevronRight className="w-4 h-4" />
   477→              </Button>
   478→            </div>
   479→          </div>
   480→
   481→          <Card className="p-4">
   482→            {calendarGrid ? (
   483→              <div>
   484→                <div className="grid grid-cols-7 gap-1 mb-1">
   485→                  {WEEKDAYS.map((d) => (
   486→                    <div key={d} className="text-center text-[10px] font-semibold text-muted-foreground py-1 uppercase tracking-wide">
   487→                      {d}
   488→                    </div>
   489→                  ))}
   490→                </div>
   491→                <div className="grid grid-cols-7 gap-1">
   492→                  {calendarGrid.padded.map((date, i) => {
   493→                    if (!date) return <div key={i} />;
   494→                    const ds = format(date, "yyyy-MM-dd");
   495→                    const dayData = calendarGrid.dayMap.get(ds);
   496→                    const ratio = dayData?.ratio ?? 0;
   497→                    const scheduled = dayData?.scheduled ?? 0;
   498→                    return (
   499→                      <Popover key={ds}>
   500→                        <PopoverTrigger asChild>
   501→                          <button
   502→                            className={cn(
   503→                              "aspect-square rounded-md text-xs flex items-center justify-center transition-all hover:scale-110 hover:ring-2 hover:ring-emerald-400 hover:ring-offset-1 hover:ring-offset-background hover:z-10 relative",
   504→                              intensityClass(ratio, scheduled),
   505→                              isToday(date) && "ring-2 ring-emerald-500 ring-offset-1 ring-offset-background",
   506→                            )}
   507→                            aria-label={`${ds}: ${dayData?.completed ?? 0} of ${scheduled} done`}
   508→                          >
   509→                            <span className={cn("font-medium", ratio > 0.4 ? "text-white" : "text-foreground")}>
   510→                              {date.getDate()}
   511→                            </span>
   512→                          </button>
   513→                        </PopoverTrigger>
   514→                        <PopoverContent className="w-64 p-3" align="center">
   515→                          <div className="text-sm font-semibold mb-1">{format(date, "EEEE, MMM d")}</div>
   516→                          {scheduled === 0 ? (
   517→                            <p className="text-xs text-muted-foreground">No habits scheduled.</p>
   518→                          ) : (
   519→                            <>
   520→                              <div className="flex items-center justify-between mb-2">
   521→                                <p className="text-xs text-muted-foreground">
   522→                                  {dayData?.completed ?? 0} of {scheduled} completed
   523→                                </p>
   524→                                <Badge variant="secondary" className="text-[10px]">
   525→                                  {Math.round(ratio * 100)}%
   526→                                </Badge>
   527→                              </div>
   528→                              <div className="space-y-1.5">
   529→                                {dayData?.habits.map((h) => (
   530→                                  <div key={h.habitId} className="flex items-center gap-2 text-xs">
   531→                                    <span className="w-5 text-center">{h.icon}</span>
   532→                                    <span className="flex-1 truncate">{h.name}</span>
   533→                                    <span
   534→                                      className={cn(
   535→                                        "px-1.5 py-0.5 rounded text-[10px] font-medium",
   536→                                        h.completed ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900 dark:text-emerald-300" : "bg-muted text-muted-foreground",
   537→                                      )}
   538→                                    >
   539→                                      {h.count}/{h.targetCount}
   540→                                    </span>
   541→                                  </div>
   542→                                ))}
   543→                              </div>
   544→                            </>
   545→                          )}
   546→                        </PopoverContent>
   547→                      </Popover>
   548→                    );
   549→                  })}
   550→                </div>
   551→                <div className="flex items-center justify-between mt-3">
   552→                  <button
   553→                    onClick={() => go({ name: "history" })}
   554→                    className="text-xs text-muted-foreground hover:text-foreground transition-colors flex items-center gap-1"
   555→                  >
   556→                    View full history <ArrowRight className="w-3 h-3" />
   557→                  </button>
   558→                  <div className="flex items-center gap-1 text-[10px] text-muted-foreground">
   559→                    <span>Less</span>
   560→                    <div className="w-3 h-3 rounded bg-muted/60" />
   561→                    <div className="w-3 h-3 rounded bg-emerald-200 dark:bg-emerald-900" />
   562→                    <div className="w-3 h-3 rounded bg-emerald-400 dark:bg-emerald-700" />
   563→                    <div className="w-3 h-3 rounded bg-emerald-600 dark:bg-emerald-500" />
   564→                    <span>More</span>
   565→                  </div>
   566→                </div>
   567→              </div>
   568→            ) : (
   569→              <Skeleton className="h-64 w-full" />
   570→            )}
   571→          </Card>
   572→        </div>
   573→      </div>
   574→
   575→      {/* Keyboard shortcuts dialog */}
   576→      <Dialog open={showShortcuts} onOpenChange={setShowShortcuts}>
   577→        <DialogContent className="sm:max-w-md">
   578→          <DialogHeader>
   579→            <DialogTitle className="flex items-center gap-2">
   580→              <Keyboard className="w-5 h-5 text-emerald-500" />
   581→              Keyboard shortcuts
   582→            </DialogTitle>
   583→          </DialogHeader>
   584→          <div className="space-y-2 py-2">
   585→            <ShortcutRow keys={["1", "…", "9"]} desc="Toggle the Nth today's habit" />
   586→            <ShortcutRow keys={["?"]} desc="Open this help dialog" />
   587→            <ShortcutRow keys={["Esc"]} desc="Close dialogs" />
   588→          </div>
   589→          <div className="text-xs text-muted-foreground bg-muted/50 rounded-lg p-3">
   590→            Shortcuts only work on the dashboard when no input is focused.
   591→          </div>
   592→          <DialogFooter>
   593→            <Button onClick={() => setShowShortcuts(false)}>Got it</Button>
   594→          </DialogFooter>
   595→        </DialogContent>
   596→      </Dialog>
   597→
   598→      {/* Note dialog */}
   599→      <Dialog open={!!noteDialog} onOpenChange={(o) => !o && setNoteDialog(null)}>
   600→        <DialogContent className="sm:max-w-md">
   601→          <DialogHeader>
   602→            <DialogTitle className="flex items-center gap-2">
   603→              <StickyNote className="w-5 h-5 text-amber-500" />
   604→              Note for {noteDialog?.habitName}
   605→            </DialogTitle>
   606→          </DialogHeader>
   607→          <div className="space-y-2 py-2">
   608→            <Label htmlFor="note">Add a note about today&apos;s check-in</Label>
   609→            <Textarea
   610→              id="note"
   611→              placeholder="How did it go? Anything to remember?"
   612→              value={noteDraft}
   613→              onChange={(e) => setNoteDraft(e.target.value)}
   614→              rows={4}
   615→              maxLength={500}
   616→            />
   617→            <p className="text-xs text-muted-foreground">{noteDraft.length}/500 characters</p>
   618→          </div>
   619→          <DialogFooter>
   620→            <Button variant="outline" onClick={() => setNoteDialog(null)}>Cancel</Button>
   621→            <Button onClick={saveNote} disabled={checkinMut.isPending}>Save note</Button>
   622→          </DialogFooter>
   623→        </DialogContent>
   624→      </Dialog>
   625→    </div>
   626→  );
   627→}

   681→function HabitRow({
   682→  habit,
   683→  index,
   684→  onToggle,
   685→  onInc,
   686→  onDec,
   687→  onClick,
   688→  onAddNote,
   689→  onFreeze,
   690→}: {
   691→  habit: {
   692→    id: string;
   693→    name: string;
   694→    description: string;
   695→    color: string;
   696→    icon: string;
   697→    targetCount: number;
   698→    count: number;
   699→    completed: boolean;
   700→    note: string;
   701→    streak: { current: number; longest: number };
   702→  };
   703→  index: number;
   704→  today: string;
   705→  onToggle: () => void;
   706→  onInc: () => void;
   707→  onDec: () => void;
   708→  onClick: () => void;
   709→  onAddNote: () => void;
   710→  onFreeze: () => void;

   712→  const multi = habit.targetCount > 1;
   713→  const progressPct = multi ? Math.round((habit.count / habit.targetCount) * 100) : habit.completed ? 100 : 0;
   714→  return (
   715→    <Card className={cn(
   716→      "p-3 flex items-center gap-3 transition-all hover:shadow-md group",
   717→      habit.completed && "border-emerald-500/50 bg-emerald-500/5",
   718→    )}>
   719→      {/* Keyboard shortcut number badge (hidden on mobile) */}
   720→      <div className="hidden sm:flex w-5 h-5 rounded bg-muted text-muted-foreground text-[10px] font-semibold items-center justify-center flex-shrink-0" title={`Press ${index + 1} to toggle`}>
   721→        {index + 1}
   722→      </div>
   723→      <button
   724→        onClick={onClick}
   725→        className="w-10 h-10 rounded-lg flex items-center justify-center text-xl flex-shrink-0 transition-transform group-hover:scale-105"
   726→        style={{ backgroundColor: habit.color + "20" }}
   727→        aria-label={habit.name}
   728→      >
   729→        {habit.icon}
   730→      </button>
   731→      <button onClick={onClick} className="flex-1 min-w-0 text-left">
   732→        <div className="flex items-center gap-2">
   733→          <span className="font-medium text-sm truncate">{habit.name}</span>
   734→          {habit.completed && (
   735→            <CheckCircle2 className="w-3.5 h-3.5 text-emerald-500 flex-shrink-0" />
   736→          )}
   737→          {habit.note && (
   738→            <StickyNote className="w-3 h-3 text-amber-500 flex-shrink-0" aria-label="Has note" />
   739→          )}

   741→        <div className="flex items-center gap-2 mt-1">

   743→            <Badge variant="secondary" className="text-[10px] py-0 px-1.5 h-4 gap-0.5">
   744→              <Flame className="w-2.5 h-2.5 text-orange-500" />
   745→              {habit.streak.current}
   746→            </Badge>
   747→          )}
   748→          {multi && (
   749→            <div className="flex items-center gap-1.5 flex-1 max-w-32">
   750→              <Progress value={progressPct} className="h-1.5" />
   751→              <span className="text-[10px] text-muted-foreground tabular-nums whitespace-nowrap">
   752→                {habit.count}/{habit.targetCount}
   753→              </span>
   754→            </div>
   755→          )}
   756→          {habit.note && (
   757→            <span className="text-[10px] text-muted-foreground italic truncate max-w-24 hidden sm:inline">&ldquo;{habit.note}&rdquo;</span>

   759→        </div>

   762→      {/* Note button */}

   768→      {!habit.completed && (
   769→        <Button
   770→          variant="ghost"
   771→          size="icon"
   772→          className="h-8 w-8 flex-shrink-0"

   778→        </Button>
   779→      )}
   780→
   781→      {multi ? (
   782→        <div className="flex items-center gap-1">
   783→          <Button variant="outline" size="icon" className="h-8 w-8" onClick={onDec} disabled={habit.count === 0}>
   784→            <Minus className="w-3.5 h-3.5" />
   785→          </Button>

   788→            <Plus className="w-3.5 h-3.5" />
   789→          </Button>

   792→        <Button

   795→          onClick={onToggle}

   800→              <CheckCircle2 className="w-4 h-4 mr-1" /> Done
   801→            </>

   805→        </Button>

   807→    </Card>
