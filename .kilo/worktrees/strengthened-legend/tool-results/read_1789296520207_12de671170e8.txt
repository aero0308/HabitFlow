    65→    65→function ProgressRing({ pct, size = 148 }: { pct: number; size?: number }) {
    66→    66→  const stroke = 12;
    67→    67→  const r = (size - stroke) / 2;
    68→    68→  const circ = 2 * Math.PI * r;
    69→    69→  const offset = circ - (pct / 100) * circ;
    70→    70→  return (
    71→    71→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    72→    72→      <svg width={size} height={size} className="-rotate-90">
    73→    73→        <defs>
    74→    74→          <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    75→    75→            <stop offset="0%" stopColor="#10b981" />
    76→    76→            <stop offset="100%" stopColor="#0d9488" />
    77→    77→          </linearGradient>
    78→    78→        </defs>
    79→    79→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/20" />
    80→    80→        <motion.circle
    81→    81→          cx={size / 2}
    82→    82→          cy={size / 2}
    83→    83→          r={r}
    84→    84→          fill="none"
    85→    85→          stroke="url(#ringGrad)"
    86→    86→          strokeWidth={stroke}
    87→    87→          strokeDasharray={circ}
    88→    88→          strokeLinecap="round"
    89→    89→          initial={{ strokeDashoffset: circ }}
    90→    90→          animate={{ strokeDashoffset: offset }}
    91→    91→          transition={{ duration: 1.2, ease: "easeOut" }}
    92→    92→        />
    93→    93→      </svg>
    94→    94→      <div className="absolute inset-0 flex flex-col items-center justify-center">
    95→    95→        <motion.span
    96→    96→          key={pct}
    97→    97→          initial={{ scale: 0.5, opacity: 0 }}
    98→    98→          animate={{ scale: 1, opacity: 1 }}
    99→    99→          transition={{ type: "spring", stiffness: 200, damping: 15 }}
   100→   100→          className="text-4xl font-bold text-foreground tabular-nums"
   101→   101→        >
   102→   102→          {pct}%
   103→   103→        </motion.span>
   104→   104→        <span className="text-xs text-muted-foreground mt-0.5">today</span>
   105→   105→      </div>
   106→   106→    </div>
   107→   107→  );
   108→   108→}

   110→   110→function intensityClass(ratio: number, scheduled: number) {
   111→   111→  if (scheduled === 0) return "bg-muted/30";
   112→   112→  if (ratio === 0) return "bg-muted/60 dark:bg-muted/40";
   113→   113→  if (ratio < 0.34) return "bg-emerald-200 dark:bg-emerald-900";
   114→   114→  if (ratio < 0.67) return "bg-emerald-400 dark:bg-emerald-700";
   115→   115→  if (ratio < 1) return "bg-emerald-500 dark:bg-emerald-600";
   116→   116→  return "bg-emerald-600 dark:bg-emerald-500";
   117→   117→}
