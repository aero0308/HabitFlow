   108→  function resizeImage(file: File, maxSize: number): Promise<string> {
   109→    return new Promise((resolve, reject) => {
   110→      const reader = new FileReader();
   111→      reader.onload = (e) => {
   112→        const img = new Image();
   113→        img.onload = () => {
   114→          const canvas = document.createElement("canvas");
   115→          let { width, height } = img;
   116→          if (width > height) {
   117→            if (width > maxSize) {
   118→              height = Math.round((height * maxSize) / width);
   119→              width = maxSize;
   120→            }
   121→          } else {
   122→            if (height > maxSize) {
   123→              width = Math.round((width * maxSize) / height);
   124→              height = maxSize;
   125→            }
   126→          }
   127→          canvas.width = width;
   128→          canvas.height = height;
   129→          const ctx = canvas.getContext("2d");
   130→          if (!ctx) return reject(new Error("No canvas context"));
   131→          ctx.drawImage(img, 0, 0, width, height);
   132→          resolve(canvas.toDataURL("image/jpeg", 0.85));
   133→        };
   134→        img.onerror = reject;
   135→        img.src = e.target?.result as string;
   136→      };
   137→      reader.onerror = reject;
   138→      reader.readAsDataURL(file);
   139→    });
   140→  }
