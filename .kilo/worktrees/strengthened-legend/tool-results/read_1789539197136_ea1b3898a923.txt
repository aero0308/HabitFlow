   108→   108→  function resizeImage(file: File, maxSize: number): Promise<string> {
   109→   109→    return new Promise((resolve, reject) => {
   110→   110→      const reader = new FileReader();
   111→   111→      reader.onload = (e) => {
   112→   112→        const img = new Image();
   113→   113→        img.onload = () => {
   114→   114→          const canvas = document.createElement("canvas");
   115→   115→          let { width, height } = img;
   116→   116→          if (width > height) {
   117→   117→            if (width > maxSize) {
   118→   118→              height = Math.round((height * maxSize) / width);
   119→   119→              width = maxSize;
   120→   120→            }
   121→   121→          } else {
   122→   122→            if (height > maxSize) {
   123→   123→              width = Math.round((width * maxSize) / height);
   124→   124→              height = maxSize;
   125→   125→            }
   126→   126→          }
   127→   127→          canvas.width = width;
   128→   128→          canvas.height = height;
   129→   129→          const ctx = canvas.getContext("2d");
   130→   130→          if (!ctx) return reject(new Error("No canvas context"));
   131→   131→          ctx.drawImage(img, 0, 0, width, height);
   132→   132→          resolve(canvas.toDataURL("image/jpeg", 0.85));
   133→   133→        };
   134→   134→        img.onerror = reject;
   135→   135→        img.src = e.target?.result as string;
   136→   136→      };
   137→   137→      reader.onerror = reject;
   138→   138→      reader.readAsDataURL(file);
   139→   139→    });
   140→   140→  }
